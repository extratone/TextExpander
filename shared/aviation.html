tz

#! /bin/bash

/*
Enter a time using 24H. 1:30pm is 13:30
*/
ENTERTIME="(Hour):(Minute)"

timezones=( "America/Los_Angeles" "America/Chicago" "Europe/Dublin" )

for zone in ${timezones[@]}
do
TZ=$zone date -jf "%H:%M %z" "$ENTERTIME $(date "+%z")" "+%H:%M %Z";
done
